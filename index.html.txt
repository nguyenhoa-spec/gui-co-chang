    void loadPhotos() {
        File dir = new File("photos");
        File[] fs = dir.listFiles((d, n) -> n.toLowerCase().matches(".*\\.(jpg|jpeg|png|gif)"));
        if (fs == null) return;
        Arrays.sort(fs);
        for (File f : fs) {
            try { BufferedImage im = ImageIO.read(f); if (im != null) photos.add(im); } catch (Exception ignored) {}
            if (photos.size() >= 4) break;
        }
    }

    // ---------- tiện ích ----------
    static double clamp(double v) { return Math.max(0, Math.min(1, v)); }
    static double ease(double x) { x = clamp(x); return x*x*(3-2*x); }
    static double easeOutBack(double x) { x = clamp(x); double c=1.70158; return 1+(c+1)*Math.pow(x-1,3)+c*Math.pow(x-1,2); }
    static Color alpha(Color c, double a) { return new Color(c.getRed(), c.getGreen(), c.getBlue(), (int)(255*clamp(a))); }
    static void centerText(Graphics2D g, String s, double cx, double y) {
        g.drawString(s, (float)(cx - g.getFontMetrics().stringWidth(s)/2.0), (float)y);
    }

    @Override protected void paintComponent(Graphics g0) {
        super.paintComponent(g0);
        Graphics2D g = (Graphics2D) g0.create();
        int w = getWidth(), h = getHeight();
        scale = Math.min(w/(double)LW, h/(double)LH);
        offX = (w - LW*scale)/2; offY = (h - LH*scale)/2;
        g.translate(offX, offY); g.scale(scale, scale);
        g.setClip(0, 0, LW, LH);
        render(g, now());
        g.dispose();
    }

    // ---------- VẼ TOÀN CẢNH ----------
    void render(Graphics2D g, double t) {
        g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON);
        g.setRenderingHint(RenderingHints.KEY_STROKE_CONTROL, RenderingHints.VALUE_STROKE_PURE);

        drawBackground(g, t);
        drawTitle(g, t);
        drawWishes(g, t);

        double cx = 500, cy = 565;
        // ánh sáng dưới sách
        double glow = ease((t-0.3)/1.5);
        g.setPaint(new RadialGradientPaint(new Point2D.Double(cx, cy+10), 330,
                new float[]{0f, 1f}, new Color[]{new Color(120,170,255,(int)(150*glow)), new Color(120,170,255,0)}));
        g.fill(new Rectangle2D.Double(cx-330, cy-330, 660, 660));

        drawBook(g, t, cx, cy);
        drawBouquet(g, t, cx, cy - 20);
        drawSparks(g, t, cx, cy);
        drawPhotos(g, t);
        drawButton(g, t);
        drawLetter(g, t);
    }

    void drawBackground(Graphics2D g, double t) {
        g.setPaint(new GradientPaint(0, 0, new Color(10, 12, 48), 0, LH, new Color(60, 24, 110)));
        g.fillRect(0, 0, LW, LH);
        g.setPaint(new RadialGradientPaint(new Point2D.Double(LW/2.0, LH*0.78), 520,
                new float[]{0f,1f}, new Color[]{new Color(120,60,190,120), new Color(120,60,190,0)}));
        g.fillRect(0, 0, LW, LH);
        for (double[] s : stars) {
            double tw = 0.5 + 0.5*Math.sin(t*2 + s[3]);
            g.setColor(new Color(255,255,255,(int)(70+150*tw)));
            double r = s[2]*(0.7+0.5*tw);
            g.fill(new Ellipse2D.Double(s[0]-r/2, s[1]-r/2, r, r));
        }
    }

        g.setColor(new Color(255, 214, 120, (int)(40*a)));

    // ---------- lời chúc bay ----------
    void drawWishes(Graphics2D g, double t) {
        double[][] slots = {{250,175},{750,200},{250,275},{750,300}};
        double dur = 8.8, gap = 2.2, t0 = 3.0;
        g.setFont(new Font(Font.SERIF, Font.ITALIC, 19));
        for (int k = 0; k < WISHES.length; k++) {
            double tt = t - t0 - k*gap;
            if (tt < 0) continue;
            tt = tt % (dur * 2);       // hai lời chúc luân phiên mỗi ô
            if (tt > dur) continue;
            double a = Math.pow(Math.sin(Math.PI * tt / dur), 0.5);
            double[] s = slots[k % 4];
            double y = s[1] - tt*3;
            g.setColor(alpha(new Color(0,0,0), a*0.0));
            g.setColor(alpha(new Color(255, 236, 200), a));
            FontMetrics fm = g.getFontMetrics();
            List<String> ls = new ArrayList<>(); StringBuilder cur = new StringBuilder();
            for (String wd : WISHES[k].split(" ")) {
                if (cur.length() > 0 && fm.stringWidth(cur + " " + wd) > 235) { ls.add(cur.toString()); cur = new StringBuilder(); }
                if (cur.length() > 0) cur.append(' ');
                cur.append(wd);
            }
            ls.add(cur.toString());
            for (int i = 0; i < ls.size(); i++) centerText(g, ls.get(i), s[0], y + i*24);
            g.setColor(alpha(new Color(255, 160, 210), a*0.8));
            drawHeart(g, s[0], y + ls.size()*24 - 6, 5);
        }
    }

    // ---------- SÁCH ----------
    void drawBook(Graphics2D g, double t, double cx, double cy) {
        double pw = 200, ph = 130, top = cy - ph/2;
        double appear = ease(t/1.0);
        g.setComposite(AlphaComposite.SrcOver.derive((float)appear));
        double open = ease((t-1.0)/2.0) * Math.PI;

        // bóng
        g.setColor(new Color(0,0,0,90));
        g.fill(new Ellipse2D.Double(cx-pw-20, cy+ph/2-6, pw*2+40, 34));

        // bìa dưới (đỏ, rộng hơn trang)
        RoundRectangle2D cover = new RoundRectangle2D.Double(cx-pw-10, top-8, pw*2+20, ph+22, 12, 12);
        g.setPaint(new GradientPaint(0, (float)top, new Color(150,30,50), 0, (float)(top+ph), new Color(90,10,30)));
        g.fill(cover);
        g.setColor(new Color(255,210,120)); g.setStroke(new BasicStroke(2f)); g.draw(cover);

        // trang bên phải
        Rectangle2D rp = new Rectangle2D.Double(cx, top, pw, ph);
        g.setColor(new Color(255, 249, 235)); g.fill(rp);
        drawPageLines(g, cx, top, pw, ph, 1, 0.3);
        for (int i = 0; i < 4; i++) { g.setColor(new Color(210,195,165)); g.draw(new Line2D.Double(cx, top+ph+2+i, cx+pw-i*2, top+ph+2+i)); }

        // bìa lật (nắp)
        drawFlap(g, cx, top, pw, ph, open, true);

        // các trang lật thêm
        for (int i = 0; i < 3; i++) {
            double a = ease((t - (3.3 + i*0.55)) / 0.9) * Math.PI;
            if (a > 0.001) drawFlap(g, cx, top, pw, ph, a, false);
        }

        // gáy sách
        g.setColor(new Color(0,0,0,70)); g.setStroke(new BasicStroke(2f));
        g.draw(new Line2D.Double(cx, top, cx, top+ph));
        g.setComposite(AlphaComposite.SrcOver);
    }

    void drawFlap(Graphics2D g, double cx, double top, double pw, double ph, double a, boolean isCover) {
        double x2 = cx + pw*Math.cos(a);
        double lift = Math.sin(a)*20;
        Path2D p = new Path2D.Double();
        p.moveTo(cx, top); p.lineTo(x2, top-lift); p.lineTo(x2, top+ph-lift*0.6); p.lineTo(cx, top+ph); p.closePath();
        boolean front = a < Math.PI/2;
        double shade = Math.abs(Math.cos(a));
        if (isCover && front) {
            g.setPaint(new GradientPaint((float)cx, 0, new Color(190,40,60), (float)x2, 0, new Color(120,20,40)));
            g.fill(p);
            g.setColor(new Color(255,210,120)); g.setStroke(new BasicStroke(2f)); g.draw(p);
            // chữ trên bìa
            if (shade > 0.35) {
                Graphics2D gg = (Graphics2D) g.create();
                gg.translate(cx, top); gg.scale(Math.cos(a), 1);
                gg.setFont(new Font(Font.SERIF, Font.BOLD, 22));
                gg.setColor(new Color(255,222,140));
                centerText(gg, "20 / 11", pw/2, ph/2-2);
                gg.setFont(new Font(Font.SERIF, Font.ITALIC, 13));
                centerText(gg, "Nhà giáo Việt Nam", pw/2, ph/2+20);
                gg.dispose();
            }
        } else {
            int c = (int)(215 + 40*shade);
            g.setColor(new Color(Math.min(255,c), Math.min(255,c-4), Math.min(255,c-18)));
            g.fill(p);
            g.setColor(new Color(190,175,150)); g.setStroke(new BasicStroke(1f)); g.draw(p);
            Graphics2D gg = (Graphics2D) g.create();
            gg.clip(p);
            gg.translate(cx, top - lift*0.3); gg.scale(Math.cos(a), 1);
            drawPageLines(gg, 0, 0, -pw, ph, -1, 0.25);
            gg.dispose();
        }
    }

    void drawPageLines(Graphics2D g, double x, double y, double w, double h, int dir, double a) {
        g.setColor(new Color(150,120,90,(int)(255*a))); g.setStroke(new BasicStroke(1f));
        for (int i = 0; i < 7; i++) {
            double yy = y + 22 + i*(h-40)/6.0;
            double len = (i==6 ? 0.45 : 0.82) * Math.abs(w);
            double x0 = x + (w>0 ? 14 : -14);
            g.draw(new Line2D.Double(x0, yy, x0 + (w>0?len:-len), yy));
        }
    }

    // ---------- BÓ HOA ----------
    void drawBouquet(Graphics2D g, double t, double ox, double oy) {
        double p = ease((t - 2.6) / 2.0);
        if (p <= 0) return;
        double s = 0.25 + 0.95*easeOutBack((t-2.6)/2.0);
        double rise = -70 * p;
        Graphics2D b = (Graphics2D) g.create();
        b.translate(ox, oy + rise);
        b.scale(s, s);
        b.setComposite(AlphaComposite.SrcOver.derive((float)clamp(p*1.5)));
        double sway = Math.sin(t*1.4)*0.03;
        b.rotate(sway, 0, 40);

        // lá
        for (int i = 0; i < 9; i++) {
            double ang = -Math.PI*0.5 + (i-4)*0.27;
            drawLeaf(b, 0, -20, ang, 150 + (i%2)*20, new Color(40+ i*5, 150, 90));
        }
        // cành
        b.setStroke(new BasicStroke(3f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        b.setColor(new Color(60, 150, 80));
        for (int i = 0; i < flowers.length; i++) {
            double[] f = flowers[i];
            double fx = f[0] + Math.sin(t*1.6 + i)*3, fy = f[1];
            b.draw(new QuadCurve2D.Double(0, 20, f[0]*0.3, (20+fy)/2, fx, fy));
        }
        // hoa
        for (int i = 0; i < flowers.length; i++) {
            double[] f = flowers[i];
            double fx = f[0] + Math.sin(t*1.6 + i)*3;
            Color c = PAL[(int)f[4]];
            if (f[3] == 0) drawRose(b, fx, f[1], f[2], c);
            else drawDaisy(b, fx, f[1], f[2], c, t*0.3 + i);
        }
        // giấy gói
        Path2D w1 = new Path2D.Double();
        w1.moveTo(-95, -75); w1.curveTo(-80, -30, -40, 30, -14, 110); w1.lineTo(14, 110); w1.curveTo(40, 30, 80, -30, 95, -75);
        w1.curveTo(40, -40, -40, -40, -95, -75); w1.closePath();
        b.setPaint(new GradientPaint(-90, -75, new Color(255,170,200), 90, 100, new Color(220,90,150)));
        b.fill(w1);
        b.setColor(new Color(255,255,255,120)); b.setStroke(new BasicStroke(1.5f)); b.draw(w1);
        Path2D w2 = new Path2D.Double();
        w2.moveTo(-60, -52); w2.lineTo(0, 100); w2.lineTo(-14, 110); w2.lineTo(-95, -75); w2.closePath();
        b.setColor(new Color(255, 245, 250, 90)); b.fill(w2);
        // nơ
        b.setColor(new Color(255, 215, 110));
        Path2D bow = new Path2D.Double();
        bow.moveTo(0, 25); bow.curveTo(-48, -5, -50, 55, 0, 28); bow.curveTo(50, 55, 48, -5, 0, 25); bow.closePath();
        b.fill(bow);
        b.setColor(new Color(235, 180, 70));
        b.fill(new Ellipse2D.Double(-8, 18, 16, 16));
        b.setStroke(new BasicStroke(5f, BasicStroke.CAP_ROUND, 0));
        b.draw(new QuadCurve2D.Double(0, 32, -14, 60, -26, 88));
        b.draw(new QuadCurve2D.Double(0, 32, 14, 60, 26, 88));
        b.dispose();
    }

    void drawLeaf(Graphics2D g, double x, double y, double ang, double len, Color c) {
        Graphics2D l = (Graphics2D) g.create();
        l.translate(x, y); l.rotate(ang);
        Path2D p = new Path2D.Double();
        p.moveTo(0, 0); p.curveTo(len*0.3, -len*0.17, len*0.75, -len*0.12, len, 0);
        p.curveTo(len*0.75, len*0.12, len*0.3, len*0.17, 0, 0);
        l.setColor(c); l.fill(p);
        l.setColor(new Color(255,255,255,60)); l.draw(new Line2D.Double(0,0,len*0.9,0));
        l.dispose();
    }

    void drawRose(Graphics2D g, double x, double y, double r, Color c) {
        g.setColor(c.darker()); g.fill(new Ellipse2D.Double(x-r, y-r, 2*r, 2*r));
        for (int i = 0; i < 5; i++) {
            double rr = r*(1 - i*0.19);
            double off = i*0.13*r;
            g.setColor(i % 2 == 0 ? c : c.brighter());
            g.fill(new Ellipse2D.Double(x-rr+Math.sin(i)*off*0.3, y-rr+off*0.1, 2*rr, 2*rr));
            g.setColor(new Color(0,0,0,40)); g.setStroke(new BasicStroke(1.2f));
            g.draw(new Arc2D.Double(x-rr*0.8+Math.sin(i)*off*0.3, y-rr*0.8, 1.6*rr, 1.6*rr, 30+i*70, 200, Arc2D.OPEN));
        }
    }

    void drawDaisy(Graphics2D g, double x, double y, double r, Color c, double spin) {
        int n = 10;
        for (int i = 0; i < n; i++) {
            double a = spin*0.2 + i*2*Math.PI/n;
            Graphics2D p = (Graphics2D) g.create();
            p.translate(x, y); p.rotate(a);
            p.setColor(c);
            p.fill(new Ellipse2D.Double(r*0.15, -r*0.2, r*0.85, r*0.4));
            p.setColor(new Color(0,0,0,35)); p.draw(new Ellipse2D.Double(r*0.15, -r*0.2, r*0.85, r*0.4));
            p.dispose();
        }
        g.setPaint(new RadialGradientPaint(new Point2D.Double(x-r*0.05, y-r*0.05), (float)(r*0.35),
                new float[]{0f,1f}, new Color[]{new Color(255,235,120), new Color(240,150,30)}));
        g.fill(new Ellipse2D.Double(x-r*0.35, y-r*0.35, r*0.7, r*0.7));
    }

    // ---------- tia sáng bay lên từ sách ----------
    void drawSparks(Graphics2D g, double t, double cx, double cy) {
        double on = ease((t-2.0)/1.0);
        if (on <= 0) return;
        Color[] cols = { new Color(255,120,200), new Color(255,220,120), new Color(160,200,255), new Color(255,255,255) };
        for (double[] s : sparks) {
            double ph = ((t*s[1] + s[2]) % 1.0);
            double x = cx + s[3]*(0.3+ph*0.9) + Math.sin(t*2 + s[2]*20)*10;
            double y = cy + 10 - ph*430;
            double a = Math.sin(Math.PI*ph) * on;
            double r = s[4]*(1-ph*0.4);
            Color c = cols[(int)s[5]];
            g.setColor(alpha(c, a*0.35)); g.fill(new Ellipse2D.Double(x-r*1.8, y-r*1.8, r*3.6, r*3.6));
            g.setColor(alpha(c, a)); g.fill(new Ellipse2D.Double(x-r/2, y-r/2, r, r));
        }
    }

    // ---------- ảnh kỷ niệm ----------
    void drawPhotos(Graphics2D g, double t) {
        double[][] pos = {{110,430,-7},{890,430,6},{110,615,5},{890,615,-6}};
        String[] cap = {"Ngày đầu gặp Cô", "Buổi học đáng nhớ", "Cô và lớp mình", "Kỉ niệm mãi trong tim"};
        for (int i = 0; i < 4; i++) {
            double tt = (t - (4.2 + i*0.5)) / 0.8;
            if (tt <= 0) continue;
            double sc = easeOutBack(tt);
            double bob = Math.sin(t*1.2 + i*1.7)*5;
            Graphics2D p = (Graphics2D) g.create();
            p.translate(pos[i][0], pos[i][1] + bob);
            p.rotate(Math.toRadians(pos[i][2] + Math.sin(t + i)*1.2));
            p.scale(sc, sc);
            p.setComposite(AlphaComposite.SrcOver.derive((float)clamp(tt)));
            int W = 140, H = 168;
            p.setColor(new Color(0,0,0,90)); p.fill(new RoundRectangle2D.Double(-W/2+5, -H/2+7, W, H, 6, 6));
            p.setColor(new Color(255,252,245)); p.fill(new RoundRectangle2D.Double(-W/2, -H/2, W, H, 6, 6));
            Shape imgBox = new Rectangle2D.Double(-W/2+8, -H/2+8, W-16, H-48);
            Graphics2D q = (Graphics2D) p.create();
            q.clip(imgBox);
            if (i < photos.size()) {
                BufferedImage im = photos.get(i);
                double k = Math.max((W-16.0)/im.getWidth(), (H-48.0)/im.getHeight());
                double iw = im.getWidth()*k, ih = im.getHeight()*k;
                q.drawImage(im, (int)(-iw/2), (int)(-H/2+8+(H-48-ih)/2), (int)iw, (int)ih, null);
            } else {
                q.setPaint(new GradientPaint(-W/2f, -H/2f, new Color(255,170,200), W/2f, H/2f, new Color(130,110,230)));
                q.fill(imgBox);
                q.setColor(new Color(255,255,255,200));
                drawHeart(q, 0, -14, 22);
                q.setFont(new Font(Font.SERIF, Font.ITALIC, 12));
                centerText(q, "Ảnh kỷ niệm " + (i+1), 0, 34);
            }
            q.dispose();
            p.setColor(new Color(90,50,70));
            p.setFont(new Font(Font.SERIF, Font.ITALIC, 12));
            centerText(p, cap[i], 0, H/2-14);
            p.dispose();
        }
    }

    void drawHeart(Graphics2D g, double x, double y, double s) {
        Path2D h = new Path2D.Double();
        h.moveTo(x, y + s*0.9);
        h.curveTo(x - s*1.6, y - s*0.1, x - s*0.7, y - s*1.2, x, y - s*0.4);
        h.curveTo(x + s*0.7, y - s*1.2, x + s*1.6, y - s*0.1, x, y + s*0.9);
        g.fill(h);
    }

    // ---------- nút mở thư ----------
    Rectangle2D buttonRect() { return new Rectangle2D.Double(LW/2.0-150, 658, 300, 46); }

    void drawButton(Graphics2D g, double t) {
        double a = ease((t-5.0)/1.0);
        if (a <= 0 || letterOpen) return;
        double pulse = 1 + 0.03*Math.sin(t*4);
        Rectangle2D r = buttonRect();
        Graphics2D b = (Graphics2D) g.create();
        b.translate(r.getCenterX(), r.getCenterY()); b.scale(pulse, pulse);
        b.setComposite(AlphaComposite.SrcOver.derive((float)a));
        RoundRectangle2D rr = new RoundRectangle2D.Double(-150, -23, 300, 46, 46, 46);
        b.setPaint(new GradientPaint(-150, 0, new Color(255,120,170), 150, 0, new Color(170,110,255)));
        b.fill(rr);
        b.setColor(new Color(255,255,255,180)); b.setStroke(new BasicStroke(1.5f)); b.draw(rr);
        // phong thư
        b.setColor(Color.WHITE);
        b.fill(new RoundRectangle2D.Double(-118, -9, 26, 18, 3, 3));
        b.setColor(new Color(200,90,140)); b.setStroke(new BasicStroke(1.4f));
        b.draw(new Line2D.Double(-118,-8,-105,2)); b.draw(new Line2D.Double(-92,-8,-105,2));
        b.setColor(Color.WHITE);
        b.setFont(new Font(Font.SERIF, Font.BOLD, 18));
        b.drawString("Mở tâm thư gửi Cô", -78, 6);
        b.dispose();
    }

    // ---------- tâm thư ----------
    void drawLetter(Graphics2D g, double t) {
        if (!letterOpen) return;
        double dt = t - letterAt;
        double dim = ease(dt/0.4);
        g.setColor(new Color(10, 5, 30, (int)(190*dim)));
        g.fillRect(0, 0, LW, LH);

        double pa = ease((dt-0.15)/0.6);
        double W = 640, H = 470, x = (LW-W)/2, y = (LH-H)/2 + (1-pa)*30;
        Graphics2D p = (Graphics2D) g.create();
        p.setComposite(AlphaComposite.SrcOver.derive((float)pa));
        p.setColor(new Color(0,0,0,100)); p.fill(new RoundRectangle2D.Double(x+6, y+8, W, H, 10, 10));
        p.setPaint(new GradientPaint((float)x, (float)y, new Color(255,252,243), (float)x, (float)(y+H), new Color(250,238,215)));
        p.fill(new RoundRectangle2D.Double(x, y, W, H, 10, 10));
        p.setColor(new Color(200,160,110)); p.setStroke(new BasicStroke(2f));
        p.draw(new RoundRectangle2D.Double(x+10, y+10, W-20, H-20, 8, 8));

        p.setFont(new Font(Font.SERIF, Font.BOLD, 28));
        p.setColor(new Color(150,30,60));
        centerText(p, LETTER_TITLE, LW/2.0, y+58);
        p.setColor(new Color(200,150,100));
        p.draw(new Line2D.Double(LW/2.0-90, y+72, LW/2.0+90, y+72));
        p.setColor(new Color(220,90,130)); drawHeart(p, LW/2.0, y+72, 6);

        Font body = new Font(Font.SERIF, Font.PLAIN, 17);
        p.setFont(body);
        FontMetrics fm = p.getFontMetrics();
        List<String> lines = new ArrayList<>();
        for (String para : LETTER) {
            StringBuilder cur = new StringBuilder();
            for (String word : para.split(" ")) {
                if (fm.stringWidth(cur + word) > W - 90) { lines.add(cur.toString().trim()); cur = new StringBuilder(); }
                cur.append(word).append(' ');
            }
            lines.add(cur.toString().trim()); lines.add("");
        }
        int shown = (int) Math.max(0, (dt-0.7) * 70);
        p.setColor(new Color(70, 40, 40));
        double ly = y + 108;
        for (String ln : lines) {
            if (shown <= 0) break;
            String s = ln.length() <= shown ? ln : ln.substring(0, shown);
            p.drawString(s, (float)(x+45), (float)ly);
            shown -= ln.length();
            ly += ln.isEmpty() ? 10 : 25;
        }
        if (shown > 0) {
            double sa = ease((dt - 0.7 - 700/70.0) / 1.0);
            p.setComposite(AlphaComposite.SrcOver.derive((float)(pa*sa)));
            p.setFont(new Font(Font.SERIF, Font.ITALIC, 20));
            p.setColor(new Color(150,30,60));
            p.drawString(SIGNATURE, (float)(x+W-45-p.getFontMetrics().stringWidth(SIGNATURE)), (float)(y+H-34));
        }
        p.dispose();

        // chớp trắng khi mở
        double fl = 1 - clamp(dt/0.7);
        if (fl > 0) { g.setColor(new Color(255,255,255,(int)(255*fl*fl))); g.fillRect(0,0,LW,LH); }

        if (dt > 1.5) {
            g.setFont(new Font(Font.SERIF, Font.ITALIC, 14));
            g.setColor(new Color(255,255,255,170));
            centerText(g, "(nhấn chuột để đóng thư)", LW/2.0, LH-18);
        }
    }

    // ---------- chuột ----------
    public void mouseClicked(MouseEvent e) {
        double lx = (e.getX() - offX)/scale, ly = (e.getY() - offY)/scale;
        double t = now();
        if (letterOpen) { if (t - letterAt > 0.8) letterOpen = false; return; }
        boolean onBtn = t > 5 && buttonRect().contains(lx, ly);
        boolean onBook = t > 3 && new Rectangle2D.Double(380, 480, 240, 150).contains(lx, ly);
        if (onBtn || onBook) { letterOpen = true; letterAt = t; }
    }

    public void mouseExited(MouseEvent e) {}
